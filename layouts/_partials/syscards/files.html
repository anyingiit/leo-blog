{{- /* 一份卡的两个下载文件：返回 dict html/pdf，各有 url、name（下载用的原文件名）、size、sha、bookmarks。 文件是这份卡页面（中文页）的资源；HTML 在内容目录里存成 .html.raw（.html 会被 Hugo 当成文章），以 .html 的名字发布到 files/ 下。 英文页用中文页的文件，不另发一份 */ -}} {{- $p := . }}{{ range .AllTranslations }}{{ if eq .Lang "zh" }}{{ $p = . }}{{ end }}{{ end }} {{- $out := dict }} {{- range $kind, $f := $p.Params.downloads }} {{- with $p.Resources.Get $f.file }} {{- $name := strings.TrimSuffix ".raw" (path.Base .Name) }} {{- with resources.Copy (path.Join (strings.TrimPrefix "/" $p.RelPermalink) "files" $name) . }} {{- $out = merge $out (dict $kind (dict "url" .RelPermalink "name" $f.name "sha" $f.sha256 "bookmarks" $f.bookmarks "size" (cond (lt $f.bytes 1048576) (printf "%.0f KB" (div (float $f.bytes) 1024)) (printf "%.1f MB" (div (float $f.bytes) 1048576))))) }} {{- end }} {{- else }} {{- errorf "系统卡下载文件不存在：%s %s" $p.File.Path $f.file }} {{- end }} {{- end }} {{- return $out }}
