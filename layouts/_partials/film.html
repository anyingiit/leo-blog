$files -}} {{- $rel := strings.TrimPrefix (printf "films/%s/" $name) (strings.TrimPrefix "/" .Name) -}} {{- if hasPrefix $rel "poster." }}{{ continue }}{{ end -}} {{- if eq $rel "index.html" -}} {{- {{- $html = replaceRE `(?i)(]*>)` (printf "${1}%s" $head) $html 1 -}} {{- $_ := (resources.FromString (printf "%sindex.html" $base) $html).RelPermalink -}} {{- else -}} {{- $_ := (resources.Copy (printf "%s%s" $base $rel) .).RelPermalink -}} {{- end -}} {{- end -}} {{- $poster := dict -}} {{- with or (resources.Get (printf "films/%s/poster.png" $name)) (resources.Get (printf -}} {{- $minutes := 0 -}} {{- $lang := "" -}} {{- with $filmPage -}} {{- $p := . -}} {{- with $ctx }}{{ range $filmPage.AllTranslations }}{{ if eq .Lang $ctx.Lang }}{{ $p = . }}{{ end }}{{ end }}{{ end -}} {{- $page = $p.RelPermalink -}} {{- $title = $p.Title -}} {{- $minutes = $p.Params.minutes | default 0 -}} {{- $lang = $p.Params.filmLang | default "" -}} {{- end -}} {{- $play := printf "/%s" $base | relURL -}} {{- return dict "play" $play "page" $page "poster" $poster.src "width" $poster.width "height" $poster.height "title" $title "minutes" $minutes "lang" $lang -}}
